"""Example source code for tokenizer evaluation."""

from __future__ import annotations
import torch
import torch.nn as nn
from typing import Protocol, Iterator
from dataclasses import dataclass

@dataclass
class Config:
    hidden_dim: int = 256
    num_layers: int = 4
    dropout: float = 0.1

class Encoder(nn.Module):
    def __init__(self, config: Config):
        super().__init__()
        self.embedding = nn.Embedding(256, config.hidden_dim)
        self.layers = nn.ModuleList([
            nn.TransformerEncoderLayer(
                d_model=config.hidden_dim,
                nhead=8,
                dim_feedforward=config.hidden_dim * 4,
                dropout=config.dropout,
                batch_first=True
            )
            for _ in range(config.num_layers)
        ])
        self.norm = nn.LayerNorm(config.hidden_dim)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        h = self.embedding(x)
        for layer in self.layers:
            h = layer(h)
        return self.norm(h)

def compute_curvature(z: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    """Estimate local curvature via finite differences."""
    batch, dim = z.shape
    curvature = torch.zeros(batch, device=z.device)

    for i in range(dim):
        perturbation = torch.zeros_like(z)
        perturbation[:, i] = eps

        z_plus = z + perturbation
        z_minus = z - perturbation

        # Second derivative approximation
        d2 = (z_plus + z_minus - 2 * z) / (eps ** 2)
        curvature += d2[:, i].abs()

    return curvature / dim

class StreamProcessor:
    def __init__(self, encoder: Encoder, chunk_size: int = 1024):
        self.encoder = encoder
        self.chunk_size = chunk_size
        self.buffer = b""

    def process(self, stream: Iterator[bytes]) -> Iterator[torch.Tensor]:
        for chunk in stream:
            self.buffer += chunk
            while len(self.buffer) >= self.chunk_size:
                data = self.buffer[:self.chunk_size]
                self.buffer = self.buffer[self.chunk_size:]

                x = torch.tensor(list(data), dtype=torch.long).unsqueeze(0)
                yield self.encoder(x)

        # Process remaining buffer
        if self.buffer:
            x = torch.tensor(list(self.buffer), dtype=torch.long).unsqueeze(0)
            yield self.encoder(x)
            self.buffer = b""

// Rust implementation of core tokenizer logic
use std::collections::HashMap;

#[derive(Debug, Clone)]
pub struct Token {
    pub id: u32,
    pub span: Vec<u8>,
    pub start: usize,
    pub end: usize,
}

pub struct Codebook {
    embeddings: Vec<Vec<f32>>,
    dim: usize,
}

impl Codebook {
    pub fn new(num_codes: usize, dim: usize) -> Self {
        let embeddings = (0..num_codes)
            .map(|_| (0..dim).map(|_| rand::random::<f32>() * 0.1).collect())
            .collect();
        Self { embeddings, dim }
    }

    pub fn quantize(&self, z: &[f32]) -> (u32, Vec<f32>) {
        let mut best_id = 0;
        let mut best_dist = f32::MAX;

        for (id, code) in self.embeddings.iter().enumerate() {
            let dist: f32 = z.iter()
                .zip(code.iter())
                .map(|(a, b)| (a - b).powi(2))
                .sum();

            if dist < best_dist {
                best_dist = dist;
                best_id = id as u32;
            }
        }

        (best_id, self.embeddings[best_id as usize].clone())
    }
}

/* C implementation for performance-critical paths */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

typedef struct {
    float* data;
    int rows;
    int cols;
} Matrix;

Matrix* matrix_create(int rows, int cols) {
    Matrix* m = malloc(sizeof(Matrix));
    m->data = calloc(rows * cols, sizeof(float));
    m->rows = rows;
    m->cols = cols;
    return m;
}

void matrix_free(Matrix* m) {
    free(m->data);
    free(m);
}

float matrix_get(Matrix* m, int i, int j) {
    return m->data[i * m->cols + j];
}

void matrix_set(Matrix* m, int i, int j, float val) {
    m->data[i * m->cols + j] = val;
}

// Contraction mapping iteration
void equilibrium_step(Matrix* z, Matrix* weights, float eta) {
    for (int i = 0; i < z->rows; i++) {
        for (int j = 0; j < z->cols; j++) {
            float z_old = matrix_get(z, i, j);
            float f_z = tanh(z_old);  // Simple nonlinearity
            float z_new = (1.0f - eta) * z_old + eta * f_z;
            matrix_set(z, i, j, z_new);
        }
    }
}

int main(int argc, char** argv) {
    Matrix* z = matrix_create(32, 64);

    // Initialize with random values
    for (int i = 0; i < z->rows * z->cols; i++) {
        z->data[i] = ((float)rand() / RAND_MAX) * 2.0f - 1.0f;

    // Run contraction iterations

}
